-- ==========================================
-- HIDDEN AUTO STEAL EGG (Runs silently)
-- ==========================================
local function StartHiddenAutoSteal()
    task.spawn(function()
        while AutoStealEnabled do
            task.wait(0.5)
            local char = Player.Character
            if not char then continue end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then continue end
            for _, obj in pairs(workspace:GetDescendants()) do
                if obj:IsA("BasePart") and (obj.Name:lower():find("egg") or obj.Name:lower():find("huevo")) then
                    local distance = (obj.Position - hrp.Position).Magnitude
                    if distance < 15 then
                        pcall(function()
                            firetouchinterest(hrp, obj, 0)
                            firetouchinterest(hrp, obj, 1)
                        end)
                    end
                end
            end
        end
    end)
end
StartHiddenAutoSteal()

-- ==========================================
-- AUTO RUN (GROUND ONLY - NO FLY)
-- ==========================================
local function AutoRunToSpawn()
    if IsRunning then
        Rayfield:Notify({Title = "Vortexs Hub", Content = "Already running!", Duration = 3})
        return
    end
    if not SavedSpawn then
        Rayfield:Notify({Title = "Vortexs Hub", Content = "Save a Spawn Point first!", Duration = 3})
        return
    end

    local char = Player.Character
    if not char then return end

    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end

    IsRunning = true
    hum.WalkSpeed = 0
    hum.JumpPower = 0

    local bv = Instance.new("BodyVelocity")
    bv.Name = "VortexsRun"
    bv.MaxForce = Vector3.new(1e5, 0, 1e5) -- Only horizontal force
    bv.Velocity = Vector3.new(0, 0, 0)
    bv.Parent = hrp

    local bg = Instance.new("BodyGyro")
    bg.Name = "VortexsGyro"
    bg.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
    bg.P = 1e4
    bg.Parent = hrp

    local targetPos = SavedSpawn.Position
    targetPos = Vector3.new(targetPos.X, hrp.Position.Y, targetPos.Z)

    local connection
    connection = game:GetService("RunService").Heartbeat:Connect(function()
        if not IsRunning or not hrp or not hrp.Parent then
            if connection then connection:Disconnect() end
            return
        end

        local currentPos = hrp.Position
        local distance = (Vector3.new(targetPos.X, currentPos.Y, targetPos.Z) - currentPos).Magnitude

        if distance < 10 then
            IsRunning = false
            bv:Destroy()
            bg:Destroy()
            hum.WalkSpeed = 16
            hum.JumpPower = 50
            if connection then connection:Disconnect() end
            Rayfield:Notify({Title = "Vortexs Hub", Content = "Arrived at Spawn!", Duration = 3})
            return
        end

        local direction = (targetPos - currentPos)
        direction = Vector3.new(direction.X, 0, direction.Z).Unit

        bv.Velocity = direction * AutoRunSpeed
        bg.CFrame = CFrame.new(hrp.Position, hrp.Position + direction)
    end)

    Rayfield:Notify({Title = "Vortexs Hub", Content = "Running to Spawn...", Duration = 3})
end

local function StopAutoRun()
    IsRunning = false
    local char = Player.Character
    if char then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hrp then
            local bv = hrp:FindFirstChild("VortexsRun")
            local bg = hrp:FindFirstChild("VortexsGyro")
            if bv then bv:Destroy() end
            if bg then bg:Destroy() end
        end
        if hum then
            hum.WalkSpeed = 16
            hum.JumpPower = 50
        end
    end
    Rayfield:Notify({Title = "Vortexs Hub", Content = "Auto Run stopped.", Duration = 3})
end
