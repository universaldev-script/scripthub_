local function make(className, properties, parent)
	local object = Instance.new(className)
	if className == "TextLabel" or className == "TextButton" or className == "TextBox" then
		pcall(function()
			object.AutoLocalize = false
		end)
	end
	for key, value in pairs(properties or {}) do
		object[key] = value
	end
	if className == "TextLabel" or className == "TextButton" or className == "TextBox" then
		pcall(function() object.AutoLocalize = false end)
	end
	object.Parent = parent
	return object
end

local function applyEnglishOnly(root)
	local function visit(obj)
		pcall(function()
			if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
				obj.AutoLocalize = false
			end
		end)
		for _, child in ipairs(obj:GetChildren()) do
			visit(child)
		end
	end
	visit(root)
end

local function installEnglishGuard(root)
	applyEnglishOnly(root)
	return root.DescendantAdded:Connect(function(obj)
		local _ = pcall(function()
			if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
				obj.AutoLocalize = false
			end
		end)
	end)
end

local function tween(object, time, properties, style, direction)
	local info = TweenInfo.new(time or 0.18, style or Enum.EasingStyle.Quint, direction or Enum.EasingDirection.Out)
	local animation = TweenService:Create(object, info, properties)
	animation:Play()
	return animation
end

local function formatNumberWithSuffix(value)
	if type(value) == "string" then
		value = tonumber(value) or 0
	end
	if value == 0 then return "0" end
	
	local suffixes = {
		{1e18, "QI"},
		{1e15, "QA"},
		{1e12, "T"},
		{1e9, "B"},
		{1e6, "M"},
		{1e3, "K"},
	}
	
	for _, pair in ipairs(suffixes) do
		local threshold = pair[1]
		local suffix = pair[2]
		if value >= threshold then
			local formatted = value / threshold
			if formatted == math.floor(formatted) then
				return tostring(math.floor(formatted)) .. suffix
			else
				return string.format("%.1f", formatted):gsub("%.0$", "") .. suffix
			end
		end
	end
	
	return tostring(value)
end

local cleanupCallbacks = {}
local cleanedUp = false
local pendingAction = nil

local function registerCleanup(callback)
	if type(callback) ~= "function" then
		return
	end
	table.insert(cleanupCallbacks, callback)
end

local function cleanupAll()
	if cleanedUp then
		return
	end
	cleanedUp = true

	for i = #cleanupCallbacks, 1, -1 do
		local callback = cleanupCallbacks[i]
		cleanupCallbacks[i] = nil
		pcall(callback)
	end
end

local function getViewport()
	local camera = workspace.CurrentCamera
	if not camera then
		return Vector2.new(1280, 720)
	end
	return camera.ViewportSize
end

local function updateGlobalScale()
	local viewport = getViewport()
	local reference = math.min(viewport.X, viewport.Y)
	local target = reference / 720
	RootScale.Scale = math.clamp(target, 0.82, 1.22)
end

	local viewportSizeConnection = workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(updateGlobalScale)
	registerCleanup(function()
		viewportSizeConnection:Disconnect()
	end)
end

local resultsScrollTouchCanvasY = nil
local resultsScrollInputConnection

local function isResultsScrollingAllowed()
	return currentTab ~= "Credits" and currentTab ~= "Saves" and ResultsScroll.Visible and ResultsScroll.ScrollingEnabled
end

local function getResultsMaxScrollY()
	return math.max(0, ResultsScroll.AbsoluteCanvasSize.Y - ResultsScroll.AbsoluteSize.Y)
end

local function setResultsScrollY(y)
	local maxY = getResultsMaxScrollY()
	ResultsScroll.CanvasPosition = Vector2.new(0, math.clamp(y or 0, 0, maxY))
end

local function isPointInsideResults(point)
	if not ResultsScroll.Visible then
		return false
	end
	local position = ResultsScroll.AbsolutePosition
	local size = ResultsScroll.AbsoluteSize
	return point.X >= position.X
		and point.X <= position.X + size.X
		and point.Y >= position.Y
		and point.Y <= position.Y + size.Y
end

local function updateResultsScrollMode()
	local enabled = currentTab ~= "Credits" and currentTab ~= "Saves"
	ResultsScroll.ScrollingEnabled = enabled
	ResultsScroll.Active = enabled
	ResultsScroll.ScrollBarThickness = enabled and 3 or 0
	ResultsScroll.CanvasPosition = Vector2.new(0, 0)
end

local function updateTabListScrollMode()
	local enabled = true
	TabList.ScrollingEnabled = enabled
	TabList.Active = enabled
	TabList.ScrollBarThickness = enabled and 3 or 0
	TabList.CanvasPosition = Vector2.new(0, 0)
end

resultsScrollWheelConnection = ResultsScroll.InputChanged:Connect(function(input)
	if not isResultsScrollingAllowed() then
		return
	end

	if input.UserInputType == Enum.UserInputType.MouseWheel then
		local delta = input.Position.Z
		if delta ~= 0 then
			setResultsScrollY(ResultsScroll.CanvasPosition.Y - (delta * ResultsScroll.AbsoluteSize.Y * 0.16))
		end
	elseif input.UserInputType == Enum.UserInputType.Touch then
		resultsScrollTouchStart = input.Position
		resultsScrollTouchCanvasY = ResultsScroll.CanvasPosition.Y
	end
end)

resultsScrollInputConnection = UserInputService.InputChanged:Connect(function(input)
	if not isResultsScrollingAllowed() then
		return
	end

	if input.UserInputType == Enum.UserInputType.Touch and resultsScrollTouchStart and resultsScrollTouchCanvasY then
		local deltaY = input.Position.Y - resultsScrollTouchStart.Y
		if math.abs(deltaY) >= 1 then
			setResultsScrollY(resultsScrollTouchCanvasY - deltaY)
		end
	elseif input.UserInputType == Enum.UserInputType.MouseWheel and isPointInsideResults(input.Position) then
		local delta = input.Position.Z
		if delta ~= 0 then
			setResultsScrollY(ResultsScroll.CanvasPosition.Y - (delta * ResultsScroll.AbsoluteSize.Y * 0.16))
		end
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch then
		resultsScrollTouchStart = nil
		resultsScrollTouchCanvasY = nil
	end
end)

-- Chave de armazenamento no writefile/readfile (executors que suportam)
local SAVE_STORAGE_KEY = "PhantomHub_StealAnEgg_Saves"
local MAX_SAVES = 5

-- Estado interno de saves
local saveSlots = {}    -- {[slotIndex] = {toggleStates = {}, numberValues = {}}}
local activeSlot = nil  -- índice do slot ativo (carregado automaticamente)

-- Funções de persistência via executor filesystem (readfile/writefile)
local function serializeSlots()
	local data = {slots = saveSlots, activeSlot = activeSlot}
	local ok, encoded = pcall(function()
		-- Serialização manual simples (sem JSON externo)
		local function encodeTable(t, depth)
			depth = depth or 0
			if depth > 6 then return "nil" end
			local parts = {}
			for k, v in pairs(t) do
				local key = type(k) == "number" and ("[" .. tostring(k) .. "]") or ('["' .. tostring(k) .. '"]')
				local val
				if type(v) == "table" then
					val = encodeTable(v, depth + 1)
				elseif type(v) == "boolean" then
					val = tostring(v)
				elseif type(v) == "string" then
					val = '"' .. v:gsub('"', '\\"') .. '"'
				elseif type(v) == "number" then
					val = tostring(v)
				else
					val = "nil"
				end
				if val ~= "nil" then
					table.insert(parts, key .. "=" .. val)
				end
			end
			return "{" .. table.concat(parts, ",") .. "}"
		end
		return encodeTable(data)
	end)
	return ok and encoded or nil
end

local function loadSlotsFromDisk()
	if typeof(readfile) ~= "function" then return end
	local ok, content = pcall(readfile, SAVE_STORAGE_KEY)
	if not ok or type(content) ~= "string" or content == "" then return end
	local fnOk, fn = pcall(loadstring, "return " .. content)
	if not fnOk or type(fn) ~= "function" then return end
	local dataOk, data = pcall(fn)
	if not dataOk or type(data) ~= "table" then return end
	if type(data.slots) == "table" then saveSlots = data.slots end
	if type(data.activeSlot) == "number" then activeSlot = data.activeSlot end
end

local function saveSlotsToDisk()
	if typeof(writefile) ~= "function" then return end
	local encoded = serializeSlots()
	if encoded then
		pcall(writefile, SAVE_STORAGE_KEY, encoded)
	end
end

-- Coleta os estados atuais de todos os toggles e number inputs de todos os datasets
local function collectCurrentState()
	local state = {toggles = {}, numbers = {}}
	local function collectFromDataset(ds)
		for _, item in ipairs(ds or {}) do
			if item.Type == "Category" then
				collectFromDataset(item.Items)
			else
				if item.Toggle then
					local key = item.Controller or item.Title or ""
					if key ~= "" then
						state.toggles[key] = toggleStates[item] == true
					end
				end
				if item.NumberInput then
					local key = item.Title or ""
					if key ~= "" then
						state.numbers[key] = item.Value or "0"
					end
				end
			end
		end
	end
	local allDatasets = {"Main", "Auto Buy", "Farm", "Config"}
	for _, dsName in ipairs(allDatasets) do
		collectFromDataset(datasets[dsName])
	end
	return state
end

-- Aplica um estado salvo nos itens do dataset
local function applyState(state)
	if type(state) ~= "table" then return end
	local toggleData = type(state.toggles) == "table" and state.toggles or {}
	local numberData = type(state.numbers) == "table" and state.numbers or {}

	local function applyToDataset(ds)
		for _, item in ipairs(ds or {}) do
			if item.Type == "Category" then
				applyToDataset(item.Items)
			else
				if item.Toggle then
					local key = item.Controller or item.Title or ""
					if key ~= "" and toggleData[key] ~= nil then
						local wasEnabled = toggleStates[item] == true
						local nowEnabled = toggleData[key] == true
						toggleStates[item] = nowEnabled
						if wasEnabled ~= nowEnabled then
							setToggleState(item, nowEnabled)
						end
					end
				end
				if item.NumberInput then
					local key = item.Title or ""
					if key ~= "" and numberData[key] ~= nil then
						item.Value = tostring(numberData[key])
						if item._TitleLabel and item._UpdateTitleFunction then
							item._UpdateTitleFunction()
						end
					end
				end
			end
		end
	end
	local allDatasets = {"Main", "Auto Buy", "Farm", "Config"}
	for _, dsName in ipairs(allDatasets) do
		applyToDataset(datasets[dsName])
	end
	renderResults()
end

-- Restaura o estado padrão (todos toggles desligados, números em 0)
local function applyDefaultState()
	local function resetDataset(ds)
		for _, item in ipairs(ds or {}) do
			if item.Type == "Category" then
				resetDataset(item.Items)
			else
				if item.Toggle then
					local wasEnabled = toggleStates[item] == true
					toggleStates[item] = false
					if wasEnabled then
						setToggleState(item, false)
					end
				end
				if item.NumberInput then
					item.Value = "0"
					if item._TitleLabel and item._UpdateTitleFunction then
						item._UpdateTitleFunction()
					end
				end
			end
		end
	end
	local allDatasets = {"Main", "Auto Buy", "Farm", "Config"}
	for _, dsName in ipairs(allDatasets) do
		resetDataset(datasets[dsName])
	end
	renderResults()
end

local function getVisibleItems(tabName, query)
	local normalized = string.lower(query or "")

	if normalized == "" then
		return datasets[tabName] or {}
	end

	-- Search globally across every tab, but only return real options.
	-- Category headers are never search results.
	local filtered = {}
	local searchOrder = {"Main", "Auto Buy", "Farm", "Config", "Credits"}
	for _, datasetName in ipairs(searchOrder) do
		for _, item in ipairs(datasets[datasetName] or {}) do
			if item.Type == "Category" then
				for _, child in ipairs(item.Items or {}) do
					local title = string.lower(child.Title or "")
					local description = string.lower(child.Description or "")
					if string.find(title, normalized, 1, true) or string.find(description, normalized, 1, true) then
						table.insert(filtered, child)
					end
				end
			else
				local title = string.lower(item.Title or "")
				local description = string.lower(item.Description or "")
				if string.find(title, normalized, 1, true) or string.find(description, normalized, 1, true) then
					table.insert(filtered, item)
				end
			end
		end
	end

	return filtered
end

local function safeRequire(moduleScript)
	if moduleScript == nil then
		return nil
	end
	local ok, result = pcall(require, moduleScript)
	if ok then
		return result
	end
	return nil
end

local function findEggTools(eggToolDisplay)
	local tools = {}
	local seen = {}

	local function scan(container)
		if not container then
			return
		end
		for _, child in ipairs(container:GetChildren()) do
			if child:IsA("Tool") and not seen[child] then
				local ok, isEgg = pcall(function()
					return eggToolDisplay and eggToolDisplay.IsEggTool and eggToolDisplay.IsEggTool(child)
				end)
				if ok and isEgg then
					seen[child] = true
					table.insert(tools, child)
				end
			end
		end
	end

	scan(LocalPlayer.Character)
	scan(LocalPlayer:FindFirstChildOfClass("Backpack"))
	return tools
end

local function getOwnPlotPlacement()
	local shared = game:GetService("ReplicatedStorage")
	local plotState = safeRequire(shared.Client:FindFirstChild("PlotState"))
	local eggRaycastModule = safeRequire(shared.Shared.Eggs:FindFirstChild("EggPlacementRaycast"))
	if not plotState or not eggRaycastModule or not eggRaycastModule.Raycast then
		return nil
	end

	local plot = plotState.ResolvePlot and plotState.ResolvePlot() or nil
	if not plot or not plot.PetArea or not plot.CenterPoint then
		return nil
	end

	local ok, raycastResult = pcall(function()
		return eggRaycastModule.Raycast(plot.PetArea, LocalPlayer.UserId)
	end)
	if not ok or raycastResult == nil or raycastResult.Instance ~= plot.PetArea then
		return nil
	end

	local relativeCFrame = plot.CenterPoint.CFrame:ToObjectSpace(CFrame.new(raycastResult.Position))
	return relativeCFrame
end

local function placeOneEgg()
	local shared = game:GetService("ReplicatedStorage")
	local eggToolDisplay = safeRequire(shared.Shared.Eggs:FindFirstChild("EggToolDisplay"))
	local eggState = safeRequire(shared.Client:FindFirstChild("EggState"))
	if not eggToolDisplay or not eggState or not eggState.PlantEgg then
		return false
	end

	local relativeCFrame = getOwnPlotPlacement()
	if not relativeCFrame then
		return false
	end

	local tools = findEggTools(eggToolDisplay)
	for _, tool in ipairs(tools) do
		local okUid, uid = pcall(function()
			return eggToolDisplay.GetToolUid and eggToolDisplay.GetToolUid(tool)
		end)
		if okUid and uid ~= nil then
			local okPlace, placed = pcall(function()
				return eggState.PlantEgg(uid, relativeCFrame)
			end)
			if okPlace and placed == true then
				local renderer = safeRequire(shared.Shared.Eggs:FindFirstChild("PlacedEggRenderer"))
				if renderer and renderer.Refresh then
					pcall(renderer.Refresh)
				end
				return true
			end
			-- A failed PlantEgg call means the current plot/egg state is not valid anymore.
			return false
		end
	end

	return false
end

local function getEquippedEggUID()
	local character = LocalPlayer.Character
	if not character then
		return nil
	end
	for _, child in ipairs(character:GetChildren()) do
		if child:IsA("Tool") and child:GetAttribute("ItemType") == "AssetEgg" then

			if typeof(uid) == "string" and uid ~= "" then
				return uid
			end
		end
	end
	return nil
end

local function eggUIDExistsInCharacterOrBackpack(uid)
	if typeof(uid) ~= "string" or uid == "" then
		return false
	end

	local function scan(container)
		if not container then
			return false
		end
		for _, child in ipairs(container:GetChildren()) do
			if child:IsA("Tool") and child:GetAttribute("ItemType") == "AssetEgg" and child:GetAttribute("UID") == uid then
				return true
			end
		end
		return false
	end

	return scan(LocalPlayer.Character) or scan(LocalPlayer:FindFirstChildOfClass("Backpack"))
end

local function getAvailableEggUID()
	local save = getSaveModule()
	if save and save.Get then
		local ok, data = pcall(save.Get)
		if ok and typeof(data) == "table" and typeof(data.EggInventory) == "table" then
			for uid, record in pairs(data.EggInventory) do
				if typeof(uid) == "string" and typeof(record) == "table" and record.Placement == nil then
					if eggUIDExistsInCharacterOrBackpack(uid) then
						return uid
					end
				end
			end
			for uid, record in pairs(data.EggInventory) do
				if typeof(uid) == "string" and typeof(record) == "table" and record.Placement == nil then
					return uid
				end
			end
		end
	end

	local eggToolDisplay = nil
	local shared = game:GetService("ReplicatedStorage")
	if shared.Shared and shared.Shared:FindFirstChild("Eggs") then
		eggToolDisplay = safeRequire(shared.Shared.Eggs:FindFirstChild("EggToolDisplay"))
	end
	for _, tool in ipairs(findEggTools(eggToolDisplay)) do
		local ok, uid = pcall(function()
			return eggToolDisplay and eggToolDisplay.GetToolUid and eggToolDisplay.GetToolUid(tool)
		end)
		if ok and typeof(uid) == "string" and uid ~= "" then
			return uid
		end
	end

	return nil
end

local function equipOneAvailableEgg(preferredUID)
	local equipped = getEquippedEggUID()

		return eggState.WearEggTool(uid)
	end)
	if ok and equippedOk then

	end
	return nil
end

local function openReadyEggs()
	local eggState, eggRecords, renderer = getOpenEggDependencies()
	if not eggState or not eggState.ReadOwnerEggs or not eggRecords
		or not eggRecords.GrowthAlpha or not eggRecords.CurrentNightCredit
		or not renderer or not renderer.ActivateLocalEgg then
		return false
	end

	local ok, ownerEggs = pcall(function()
		return eggState.ReadOwnerEggs(LocalPlayer.UserId)
	end)
	if not ok or typeof(ownerEggs) ~= "table" then
		return false
	end

	local serverTime = workspace:GetServerTimeNow()
	local opened = false

	for uid, record in pairs(ownerEggs) do
		if typeof(uid) == "string" and typeof(record) == "table" and record.Placement then
			local ready = pcall(function()
				return eggRecords.GrowthAlpha(
					record,
					serverTime,
					record.GrowthSpeedMultiplier,
					eggRecords.CurrentNightCredit(record, serverTime, record.GrowthSpeedMultiplier),
					LocalPlayer
				) >= 1
			end)

			if ready then
				local activated, success = pcall(function()
					return renderer.ActivateLocalEgg(uid)
				end)
				if activated and success == true then
					opened = true
					task.wait(0.15)
				end
			end
		end
	end

	return opened
end

				pcall(baseUpgrade.PurchaseNextTier)
			end
			task.wait(1)
		end
	end)
end

local function getTrailRank(config)
	local speed = config and config.SpeedMultiplier
	local price = config and config.Price
	if typeof(speed) ~= "number" then speed = -math.huge end
	if typeof(price) ~= "number" then price = -math.huge end
	return speed, price
end

local function setBuyTrails(item, enabled)
	disconnectToggle(item)
	if not enabled then
		return
	end

	local remotes, trails, save = getTrailModules()
	if not remotes or not remotes.Trailwear or not remotes.Trailwear.AskPurchase or not trails or not trails.Directory or not save or not save.Get then
		return
	end

	local running = true
	toggleConnections[item] = {
		Disconnect = function()
			running = false
		end,
	}

	task.spawn(function()
		while running and not cleanedUp do
			local okData, data = pcall(save.Get)
			if okData and typeof(data) == "table" then
				local money = typeof(data.Money) == "number" and data.Money or 0
				local owned = typeof(data.TrailInventory) == "table" and data.TrailInventory or {}
				local candidates = {}

				for id, config in pairs(trails.Directory) do
					if typeof(id) == "string" and typeof(config) == "table" and config.DisplayInShop and owned[id] ~= true then
						table.insert(candidates, {Id = id, Config = config})
					end
				end

				table.sort(candidates, function(a, b)
					local aSpeed, aPrice = getTrailRank(a.Config)
					local bSpeed, bPrice = getTrailRank(b.Config)
					if aSpeed ~= bSpeed then return aSpeed > bSpeed end
					if aPrice ~= bPrice then return aPrice > bPrice end
					return a.Id < b.Id
				end)

				-- Purchase only the single highest-priority affordable trail this cycle.
				-- The loop then refreshes the inventory/money and continues with the next best trail.
				for _, entry in ipairs(candidates) do
					local price = entry.Config.Price
					if typeof(price) == "number" and price >= 0 and money >= price then
						pcall(function()
							remotes.Trailwear.AskPurchase:InvokeServer(entry.Id)
						end)
						break
					end
				end
			end

			task.wait(1)
		end
	end)
end

local function setEquipBestTrail(item, enabled)

		return
	end

	local remotes, trails, save = getTrailModules()
	if not remotes or not remotes.Trailwear or not remotes.Trailwear.AskChoose or not trails or not trails.Directory or not save or not save.Get then
		return
	end

	local running = true
	toggleConnections[item] = {
		Disconnect = function()
			running = false
		end,
	}

	task.spawn(function()
		while running and not cleanedUp do
			local okData, data = pcall(save.Get)
			if okData and typeof(data) == "table" and typeof(data.TrailInventory) == "table" then
				local bestId, bestConfig = nil, nil
				for id, owned in pairs(data.TrailInventory) do
					if owned == true and trails.Directory[id] then
						local config = trails.Directory[id]
						if not bestConfig then
							bestId, bestConfig = id, config
						else
							local speed, price = getTrailRank(config)
							local bestSpeed, bestPrice = getTrailRank(bestConfig)
							if speed > bestSpeed or (speed == bestSpeed and (price > bestPrice or (price == bestPrice and id < bestId))) then
								bestId, bestConfig = id, config
							end
						end
					end
				end
				if bestId and data.EquippedTrail ~= bestId then
					pcall(function()
						remotes.Trailwear.AskChoose:InvokeServer(bestId)
					end)
				end
			end
			task.wait(1)
		end
	end)
end

local function isBatTool(tool)
	if not tool or not tool:IsA("Tool") then
		return false
	end

	local name = string.lower(tool.Name or "")
	return string.find(name, "bat", 1, true) ~= nil
		or string.find(name, "bastão", 1, true) ~= nil
		or string.find(name, "bastao", 1, true) ~= nil
		or string.find(name, "volcano", 1, true) ~= nil
end

local function findBatTool()
	local character = LocalPlayer.Character

local function getGuardianModelFromArea(area)
	if not area or not area:IsA("Model") then
		return nil
	end

	-- The supplied GuardAreas client script resolves each area's actual guard
	-- through GuardComponent:GetGuardModel(). We mirror that result as closely
	-- as possible without instantiating a second GuardComponent.
	local directGuard = area:FindFirstChild("Guard")
	if directGuard and directGuard:IsA("Model") then
		local humanoid = directGuard:FindFirstChildOfClass("Humanoid")
		local root = directGuard:FindFirstChild("HumanoidRootPart") or directGuard:FindFirstChild("Head")
		if humanoid and root then
			return directGuard
		end
	end

	for _, child in ipairs(area:GetChildren()) do
		if child:IsA("Model") then
			local humanoid = child:FindFirstChildOfClass("Humanoid")
			local root = child:FindFirstChild("HumanoidRootPart") or child:FindFirstChild("Head")
			if humanoid and root then
				return child
			end
		end
	end

	for _, descendant in ipairs(area:GetDescendants()) do
		if descendant:IsA("Model") then
			local humanoid = descendant:FindFirstChildOfClass("Humanoid")
			local root = descendant:FindFirstChild("HumanoidRootPart") or descendant:FindFirstChild("Head")
			if humanoid and root then
				return descendant
			end
		end
	end

	return nil
end

local function getGuardianDisplayName(guardModel, area)
	if not area then
		return "Guardian"
	end

	-- Use the guard-area model name supplied by the game's GuardAreas system
	-- instead of the humanoid/model name. Example: Forest -> "Guardian of the Forest".
	local areaName = area.Name
	if type(areaName) == "string" and areaName ~= "" then
		return "Guardian of the " .. areaName
	end

	return "Guardian"
end

local function getEggDisplayName(record, model)
	-- Prefer the game's shared asset configuration for the actual egg name.
	local shared = game:GetService("ReplicatedStorage")
	local assetsModule = shared:FindFirstChild("Data") and shared.Data:FindFirstChild("Assets")
	local assets = safeRequire(assetsModule)
	local asset = nil

	if assets and type(assets) == "table" and type(assets.Directory) == "table" and record then
		asset = assets.Directory[record.AssetCategory]
	end

	if type(asset) == "table" then
		local candidates = {
			asset.DisplayName,
			asset.Name,
			asset.Title,
			asset.Display,
			asset.AssetName,
		}
		for _, value in ipairs(candidates) do
			if type(value) == "string" and value ~= "" then
				return value
			end
		end
	end

	if model then
		local candidates = {
			model:GetAttribute("DisplayName"),
			model:GetAttribute("Name"),
			model:GetAttribute("EggName"),
		}
		for _, value in ipairs(candidates) do
			if type(value) == "string" and value ~= "" then
				return value
			end
		end
	end

	if record and type(record.AssetCategory) == "string" and record.AssetCategory ~= "" then
		return record.AssetCategory
	end

	return "Egg"
end

local function getEggRarityInfo(record)
	local shared = game:GetService("ReplicatedStorage")
	local assetsModule = shared:FindFirstChild("Data") and shared.Data:FindFirstChild("Assets")
	local assets = safeRequire(assetsModule)

	if not assets or type(assets) ~= "table" or type(assets.Directory) ~= "table" or not record then
		return nil, Color3.fromRGB(255, 255, 255)
	end

	local asset = assets.Directory[record.AssetCategory]
	if type(asset) ~= "table" or type(asset.Rarity) ~= "table" then
		return nil, Color3.fromRGB(255, 255, 255)
	end

	local rarity = asset.Rarity
	local rarityName = rarity.DisplayName or rarity.Name or rarity.Title
	if type(rarityName) ~= "string" or rarityName == "" then
		rarityName = record.Rarity or record.RarityName
	end

	if type(rarityName) ~= "string" or rarityName == "" then
		rarityName = "Unknown"
	end

	local rarityColor = rarity.Color
	if typeof(rarityColor) ~= "Color3" then
		rarityColor = Color3.fromRGB(255, 255, 255)
	end

	return rarityName, rarityColor
end

local function getEggMutationName(record)
	if not record then
		return nil
	end

	local function extractMutation(value)
		if type(value) == "string" then
			if value ~= "" then
				return value
			end
			return nil
		end

		if type(value) == "table" then
			local directCandidates = {
				value.DisplayName,
				value.Name,
				value.Title,
				value.Mutation,
				value.Type,
				value.Id,
			}

			for _, candidate in ipairs(directCandidates) do
				if type(candidate) == "string" and candidate ~= "" then
					return candidate
				end
			end

			for key, candidate in pairs(value) do
				if type(candidate) == "string" and candidate ~= "" then
					if key == "Name" or key == "DisplayName" or key == "Title" or key == "Mutation" or key == "Type" or key == "Id" then
						return candidate
					end
				elseif candidate == true and type(key) == "string" and key ~= "" then
					return key
				end
			end

			for _, candidate in ipairs(value) do
				local extracted = extractMutation(candidate)
				if extracted then
					return extracted
				end
			end
		end

		return nil
	end

	local mutation = extractMutation(record.Mutations)
	if mutation then
		return mutation
	end

	return extractMutation(record.BaseMutation)
end

local function setEggESP(item, enabled)
	disconnectToggle(item)

	if not enabled then
		if refreshEggESP then
			pcall(refreshEggESP, true)
		end
		return
	end

	local running = true
	local labels = {}
	local eggState = getEggESPStateModule()

	local function destroyLabel(model)
		local label = labels[model]
		if label then
			labels[model] = nil
			pcall(function()
				label:Destroy()
			end)
		end
	end

	local function destroyAllLabels()
		for model, label in pairs(labels) do
			labels[model] = nil
			pcall(function()
				label:Destroy()
			end)
		end
	end

	local function getEggAdornee(model)
		return model.PrimaryPart
			or model:FindFirstChild("HumanoidRootPart")
			or model:FindFirstChildWhichIsA("BasePart")
	end

	local function clearInfoLabels(label)
		for _, child in ipairs(label:GetChildren()) do
			if child:IsA("TextLabel") then
				child:Destroy()
			end
		end
	end

	local function addInfoLine(label, name, text, textColor, order)
		local line = Instance.new("TextLabel")
		line.Name = name
		line.Size = UDim2.fromScale(1, 0)
		line.AutomaticSize = Enum.AutomaticSize.Y
		line.BackgroundTransparency = 1
		line.BorderSizePixel = 0
		line.Text = name .. ": " .. text
		line.TextColor3 = textColor or Color3.fromRGB(255, 255, 255)
		line.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
		line.TextStrokeTransparency = 0.30
		line.Font = Enum.Font.GothamSemibold
		line.TextSize = 10
		line.TextScaled = false
		line.TextWrapped = false
		line.TextXAlignment = Enum.TextXAlignment.Center
		line.TextYAlignment = Enum.TextYAlignment.Center
		line.LayoutOrder = order
		line.ZIndex = 2
		line.Parent = label
	end

	local function createOrUpdateLabel(model, record)
		if not model or not model.Parent then
			return
		end

		local hasAnySelection = visualSelectionState.Name
			or visualSelectionState.Distance
			or visualSelectionState.Mutation
			or visualSelectionState.Rarity

		if not hasAnySelection then
			destroyLabel(model)
			return
		end

		local adornee = getEggAdornee(model)
		if not adornee or not adornee:IsA("BasePart") then
			destroyLabel(model)
			return
		end

		local label = labels[model]
		if not label or not label.Parent then
			label = Instance.new("BillboardGui")
			label.Name = "PhantomHub_EggESP"
			label.Adornee = adornee
			label.AlwaysOnTop = true
			label.LightInfluence = 0
			label.MaxDistance = 1000
			label.Size = UDim2.fromOffset(150, 80)
			label.StudsOffsetWorldSpace = Vector3.new(0, 2.05, 0)
			label.ResetOnSpawn = false

			local layout = Instance.new("UIListLayout")
			layout.Name = "InfoLayout"
			layout.FillDirection = Enum.FillDirection.Vertical
			layout.HorizontalAlignment = Enum.HorizontalAlignment.Center
			layout.VerticalAlignment = Enum.VerticalAlignment.Center
			layout.SortOrder = Enum.SortOrder.LayoutOrder
			layout.Padding = UDim.new(0, 0)
			layout.Parent = label

			label.Parent = PlayerGui
			labels[model] = label
		end

		label.Adornee = adornee
		clearInfoLabels(label)

		local order = 1

		if visualSelectionState.Name then
			addInfoLine(label, "Name", getEggDisplayName(record, model), Color3.fromRGB(255, 255, 255), order)
			order += 1
		end

		if visualSelectionState.Distance then
			local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
			local rootPart = root and root:IsA("BasePart") and root or nil
			if not rootPart then
				local character = LocalPlayer.Character
				if character then
					rootPart = character:FindFirstChildWhichIsA("BasePart")
				end
			end

			if rootPart then
				local distance = (rootPart.Position - adornee.Position).Magnitude
				local roundedDistance = math.floor(distance + 0.5)
				addInfoLine(label, "Distance", tostring(roundedDistance) .. " studs", Color3.fromRGB(255, 255, 255), order)
			else
				addInfoLine(label, "Distance", "-- studs", Color3.fromRGB(255, 255, 255), order)
			end
			order += 1
		end

		if visualSelectionState.Mutation then
			local mutationName = getEggMutationName(record)
			if mutationName then
				addInfoLine(label, "Mutation", mutationName, Color3.fromRGB(255, 255, 255), order)
				order += 1
			end
		end

		if visualSelectionState.Rarity then
			local rarityName, rarityColor = getEggRarityInfo(record)
			addInfoLine(label, "Rarity", rarityName or "Unknown", rarityColor, order)
			order += 1
		end
	end

	local function refresh(forceDestroy)
		if forceDestroy then
			destroyAllLabels()
			return
		end

		local hasAnySelection = visualSelectionState.Name
			or visualSelectionState.Distance
			or visualSelectionState.Mutation
			or visualSelectionState.Rarity

		if not hasAnySelection then
			destroyAllLabels()
			return
		end

		local folder = workspace:FindFirstChild("AreaEggSlotsClient")
		if not folder then
			destroyAllLabels()
			return
		end

		local seen = {}

		for _, model in ipairs(folder:GetChildren()) do
			if model:IsA("Model") then
				local record = nil

				if eggState and eggState.ReadFieldEgg then
					pcall(function()
						record = eggState.ReadFieldEgg(model.Name)
					end)
				end

				if record then
					seen[model] = true
					createOrUpdateLabel(model, record)
				end
			end
		end

		for model in pairs(labels) do
			if not seen[model] or not model.Parent then
				destroyLabel(model)
			end
		end
	end

	refreshEggESP = function(forceDestroy)
		if not running then
			return
		end
		refresh(forceDestroy)
	end

	toggleConnections[item] = {
		Disconnect = function()
			running = false
			if refreshEggESP then
				refreshEggESP = nil
			end
			destroyAllLabels()
		end,
	}

	task.spawn(function()
		refresh()
		while running and not cleanedUp do
			task.wait(0.20)
			if running and not cleanedUp then
				refresh()
			end
		end
		destroyAllLabels()
	end)
end

renderResults = function()
	local previousScrollY = ResultsScroll.CanvasPosition.Y
	clearResults()

	-- Aba Saves tem renderização própria
	if currentTab == "Saves" then
		updateResultsScrollMode()
		renderSavesTab()
		return
	end

	local query = SearchBox.Text
	local items = getVisibleItems(currentTab, query)
	if #items == 0 then
		if currentTab == "Main" and query == "" then
			setResultsScrollPositionY(0)
			return
		elseif query ~= "" then
			showEmptyState("No matching options found.")
		elseif currentTab == "Auto Buy" then
			setResultsScrollPositionY(0)
			return
		elseif currentTab == "Config" then
			showEmptyState("No options are available in Config.")
		elseif currentTab == "Credits" then
			showEmptyState("No options are available in Credits.")
		else
			showEmptyState("This tab has no options yet.")
		end
		setResultsScrollPositionY(0)
		return
	end

	local layoutIndex = 0
	if query ~= "" then
		-- Global search results are option-only; categories are never rendered.
		for _, item in ipairs(items) do
			layoutIndex += 1
			createStandardRow(item, layoutIndex)
		end
		task.defer(function()
			setResultsScrollPositionY(previousScrollY)
		end)
		return
	end

	for _, item in ipairs(items) do
		layoutIndex += 1
		if item.Type == "Category" then
			createCategoryRow(item, layoutIndex, query)
			if item.Expanded then
				layoutIndex += #(item.Items or {})
			end
		else
			createStandardRow(item, layoutIndex)
		end
	end

	task.defer(function()
		setResultsScrollPositionY(previousScrollY)
		applyEnglishOnly(ResultsScroll)
	end)
	applyEnglishOnly(ResultsScroll)
	-- Garantir que o canvas size seja recalculado corretamente para melhor scrolling
	task.wait()
	local listLayout = ResultsScroll:FindFirstChildOfClass("UIListLayout")
	if listLayout then
		local canvasYSize = listLayout.AbsoluteContentSize.Y
		if canvasYSize > 0 then
			ResultsScroll.CanvasSize = UDim2.fromScale(1, math.max(1, (canvasYSize + 10) / ResultsScroll.AbsoluteSize.Y))
		end
	end
end

local function setTab(name)
	-- Tab switching remains locked while the Visual settings panel is open.
	-- Other menu interactions remain available.
	if visualPanelOpen then
		return
	end
	currentTab = name
	updateResultsScrollMode()
	for tabName, button in pairs(tabButtons) do
		local selected = tabName == name
		tween(button, 0.14, {
			BackgroundColor3 = selected and COLORS.Gray2 or COLORS.Black2,
			BackgroundTransparency = selected and 0 or 0.15,
		})
		local icon = button:FindFirstChild("Icon")
		if icon and icon:IsA("ImageLabel") then
			tween(icon, 0.14, {ImageColor3 = selected and COLORS.White or COLORS.SoftWhite})
		end
		local label = button:FindFirstChild("Label")
		if label and label:IsA("TextLabel") then
			tween(label, 0.14, {TextColor3 = selected and COLORS.White or COLORS.SoftWhite})
		end
	end
	SearchBox.Text = ""
	SearchHint.Visible = true
	SearchClear.Visible = false
	renderResults()
end

	local position = guiObject.AbsolutePosition

		and point.Y >= position.Y and point.Y <= position.Y + size.Y

local launcherDragging = false
local launcherDragInput = nil
local launcherDragStart = nil
local launcherStartPos = nil
local launcherDidDrag = false

local function clampLauncher(position)
	local viewport = getViewport()
	local x = math.clamp(position.X.Scale, 0.0, 0.925)
	local y = math.clamp(position.Y.Scale, 0.02, 0.92)

	if position.X.Scale < 0 then
		x = 0.015
	elseif position.X.Scale > 0.925 then
		x = 0.907
	end

	return UDim2.fromScale(x, y)
end

local function updateLauncherDrag(input)
	if lockEnabled and lockButtonSelected then
		return
	end
	local delta = input.Position - launcherDragStart
	if delta.Magnitude > 4 then
		launcherDidDrag = true
	end
	local viewport = getViewport()
	if viewport.X <= 0 or viewport.Y <= 0 then
		return
	end
	local dx = delta.X / viewport.X
	local dy = delta.Y / viewport.Y
	local target = UDim2.fromScale(
		launcherStartPos.X.Scale + dx,
		launcherStartPos.Y.Scale + dy
	)
	LauncherFrame.Position = clampLauncher(target)
end

local function beginLauncherDrag(input)
	if lockEnabled and lockButtonSelected then
		return
	end
	launcherDragging = true
	launcherDidDrag = false
	launcherDragStart = input.Position
	launcherStartPos = LauncherFrame.Position
	launcherDragInput = input.UserInputType == Enum.UserInputType.Touch and input or nil
end

local function endLauncherDrag()
	launcherDragging = false
	launcherDragInput = nil
	launcherDragStart = nil
	launcherStartPos = nil
end

Launcher.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		beginLauncherDrag(input)
	end
end)

			return
		end
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			pointerDownAt = os.clock()
			didDrag = false
			menuDragging = false
			menuDragInput = input.UserInputType == Enum.UserInputType.Touch and input or nil
			menuDragStart = input.Position
			menuStartPos = Menu.Position
			dragCorner = name
		end
	end)
end

function updateMenuDrag(input)
	if lockEnabled and lockMenuSelected then
		return
	end
	if not menuDragStart or not menuStartPos then
		return
	end
	local delta = input.Position - menuDragStart
	if delta.Magnitude < 4 then
		return
	end
	if os.clock() - pointerDownAt < 0.06 then
		return
	end
	didDrag = true
	menuDragging = true

	local viewport = getViewport()
	if viewport.X <= 0 or viewport.Y <= 0 then
		return
	end
	local dx = delta.X / viewport.X
	local dy = delta.Y / viewport.Y
	local x = menuStartPos.X.Scale + dx
	local y = menuStartPos.Y.Scale + dy
	Menu.Position = UDim2.fromScale(x, y)
end

function endMenuDrag()
	if not menuDragging then
		menuDragInput = nil
		return
	end

	local viewport = getViewport()
	local absolutePosition = Menu.AbsolutePosition
	local absoluteSize = Menu.AbsoluteSize
	local fullyLeft = absolutePosition.X + absoluteSize.X < 0
	local fullyRight = absolutePosition.X > viewport.X
	local fullyTop = absolutePosition.Y + absoluteSize.Y < 0
	local fullyBottom = absolutePosition.Y > viewport.Y

	if fullyLeft then
		local y = math.clamp(Menu.Position.Y.Scale, 0.05, 0.95)
		tween(Menu, 0.20, {Position = UDim2.fromScale(0.02, y)})
	elseif fullyRight then
		local y = math.clamp(Menu.Position.Y.Scale, 0.05, 0.95)
		tween(Menu, 0.20, {Position = UDim2.fromScale(0.98, y)})
	elseif fullyTop then
		tween(Menu, 0.20, {Position = UDim2.fromScale(math.clamp(Menu.Position.X.Scale, 0.02, 0.98), 0.05)})
	elseif fullyBottom then
		tween(Menu, 0.20, {Position = UDim2.fromScale(math.clamp(Menu.Position.X.Scale, 0.02, 0.98), 0.95)})
	end

	menuDragging = false
	menuDragInput = nil
	menuDragStart = nil
	menuStartPos = nil
	tween(MenuStroke, 0.12, {Color = COLORS.Gray3, Transparency = 0.12})
end

for name, corner in pairs(corners) do
	corner.Activated:Connect(function()
		if didDrag then
			didDrag = false
			return
		end
	end)
end

local inputChangedConnection = UserInputService.InputChanged:Connect(function(input)
	if launcherDragging and ((launcherDragInput and input == launcherDragInput) or (not launcherDragInput and input.UserInputType == Enum.UserInputType.MouseMovement)) then
		updateLauncherDrag(input)
	end
	if menuDragInput and input == menuDragInput then
		updateMenuDrag(input)
	elseif menuDragging and input.UserInputType == Enum.UserInputType.MouseMovement then
		updateMenuDrag(input)
	elseif menuDragStart and not menuDragInput and input.UserInputType == Enum.UserInputType.MouseMovement then
		updateMenuDrag(input)
	end
end)

local inputEndedConnection = UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		if launcherDragging then
			endLauncherDrag()
		end
		if menuDragging then
			endMenuDrag()
		end
	end
end)

end)
